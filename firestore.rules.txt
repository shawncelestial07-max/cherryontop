rules_version = '2';

service cloud.firestore {
  match /databases/{database}/documents {
    // Signed-in cashiers can read products.
    match /products/{productId} {
      allow read: if request.auth != null;

      // Checkout may change stock only. Product creation, deletion, prices,
      // names, and other product fields cannot be changed by this POS rule.
      allow update: if request.auth != null
        && request.resource.data.diff(resource.data).affectedKeys().hasOnly(['stock'])
        && request.resource.data.stock is number
        && request.resource.data.stock >= 0;
    }

    // Signed-in users can read orders and create completed orders.
    // Existing orders cannot be edited or deleted from the client.
    match /orders/{orderId} {
      allow read, create: if request.auth != null;
      allow update, delete: if false;
    }
  }
}
